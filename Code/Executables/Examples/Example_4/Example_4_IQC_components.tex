\subsection*{Signal ordering: rows and columns}
All vectors below are column vectors. The plant has $n_z=5$ regulated outputs, $n_w=3$ exogenous inputs, and $n_u=1$ control inputs.
\[u=[u_{\mathrm S}]^\top,\qquad z_\Delta=[z_{\mathrm S},z_{\mathrm G}]^\top,\qquad w_\Delta=[w_{\mathrm S},w_{\mathrm G}]^\top=\delta(z_\Delta),\qquad z_{\mathrm p}=[x_{\mathrm S},x_{\mathrm G},\mu u_{\mathrm S}]^\top.\]
\[z=\begin{bmatrix}z_\Delta\\z_{\mathrm p}\end{bmatrix}=[z_{\mathrm S},z_{\mathrm G},x_{\mathrm S},x_{\mathrm G},\mu u_{\mathrm S}]^\top,\qquad w=\begin{bmatrix}w_\Delta\\w_{\mathrm p}\end{bmatrix}=[w_{\mathrm S},w_{\mathrm G},w_{\mathrm p}]^\top.\]
\paragraph{Primal filter.} The columns of $\Theta$ are ordered by $[z^\top,w^\top]^\top$; its rows are ordered by $[\widetilde z^\top,\widetilde w^\top]^\top$. Explicitly,
\[\begin{bmatrix}\widetilde z_\Delta\\\widetilde z_{\mathrm p}\\\widetilde w_\Delta\\\widetilde w_{\mathrm p}\end{bmatrix}=\Theta\begin{bmatrix}z_\Delta\\z_{\mathrm p}\\w_\Delta\\w_{\mathrm p}\end{bmatrix},\qquad [z^\top,w^\top]^\top=[z_{\mathrm S},z_{\mathrm G},x_{\mathrm S},x_{\mathrm G},\mu u_{\mathrm S},w_{\mathrm S},w_{\mathrm G},w_{\mathrm p}]^\top,\qquad\text{block sizes }(2,3,2,1).\]
\paragraph{Dual filter.} Dual channel positions correspond to $[w^\top,z^\top]^\top$: the dual plant output $\underline z\in\mathbb R^3$ has the channel order of $w$, and its input $\underline w\in\mathbb R^{5}$ has the channel order of $z$. Underlined signals belong to the dual system.
\[\begin{bmatrix}\widetilde{\underline z}_\Delta\\\widetilde{\underline z}_{\mathrm p}\\\widetilde{\underline w}_\Delta\\\widetilde{\underline w}_{\mathrm p}\end{bmatrix}=\mathbf D(\Theta)\begin{bmatrix}\underline z_\Delta\\\underline z_{\mathrm p}\\\underline w_\Delta\\\underline w_{\mathrm p}\end{bmatrix},\qquad\text{channel positions }[w_{\mathrm S},w_{\mathrm G},w_{\mathrm p},z_{\mathrm S},z_{\mathrm G},x_{\mathrm S},x_{\mathrm G},\mu u_{\mathrm S}]^\top,\qquad\text{block sizes }(2,1,2,3).\]
The scalar factors $H_i,M_i,W_{\mathrm p}$ act on their named channels. The uncertainty-factor row/column orders are $(z_{\mathrm S},z_{\mathrm G},w_{\mathrm S},w_{\mathrm G})$ for $\Theta_\Delta$ and $(w_{\mathrm S},w_{\mathrm G},z_{\mathrm S},z_{\mathrm G})$ for $\mathbf D(\Theta_\Delta)$. The performance-factor orders are $(z_{\mathrm p},w_{\mathrm p})$ for $\Theta_{\mathrm p}$ and $(w_{\mathrm p},z_{\mathrm p})$ for $\mathbf D(\Theta_{\mathrm p})$, with the same orders for their corresponding filtered outputs.
\paragraph{Graph and signature blocks.} In the primal graph, $(\mathcal C_z,\mathcal D_z)$ produce $\widetilde z$ (5 rows), while $(\mathcal C_w,\mathcal D_w)$ produce $\widetilde w$ (3 rows); both feedthrough blocks have input $w$ (3 columns). In the dual graph, $(\underline{\mathcal C}_z,\underline{\mathcal D}_z)$ produce $\widetilde{\underline z}$ (3 rows), while $(\underline{\mathcal C}_w,\underline{\mathcal D}_w)$ produce $\widetilde{\underline w}$ (5 rows); both feedthrough blocks have input $\underline w$ (5 columns).
The primal signature $V$ uses the primal filtered order $(\widetilde z_\Delta,\widetilde z_{\mathrm p},\widetilde w_\Delta,\widetilde w_{\mathrm p})$. The dual signatures $\underline V$ use $(\widetilde{\underline z}_\Delta,\widetilde{\underline z}_{\mathrm p},\widetilde{\underline w}_\Delta,\widetilde{\underline w}_{\mathrm p})$.
