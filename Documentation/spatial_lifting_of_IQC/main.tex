\section{Problem Statement: Spatial Lifting of a Hard IQC}

We denote the extended $L_2$ space by $L_{2,e}$ and let $\mcl{F}$
denote the set of hard factorizations. Suppose that
$\Delta:L_{2,e}\to L_{2,e}$ satisfies the hard IQC associated with a
fixed $(\Psi,V)\in\mcl{F}$,
\begin{equation}\label{eqn:temporal-IQC}
    \ip{p_T\Psi\bmat{z\\ \Delta z}}{p_TV\Psi\bmat{z\\ \Delta z}}_{L_2}\geq0,
\end{equation}
for all $T>0$ and $z\in L_{2,e}$, where
$\Psi:L_{2,e}^n\to L_{2,e}^{\tilde n}$ and
$V=V^\top\in\R^{\tilde n\times\tilde n}$.

We extend the signals over the spatial domain $[a,b]$ and define the
corresponding signal space as
\begin{equation}
    L_{\mbf{H},e}:=\left\{u:\int_0^T\int_a^b|u(t,s)|^2\,ds\,dt<\infty
    \text{ for all }T>0\right\}.
\end{equation}
The lifted uncertainty $\hat{\Delta}$ is defined pointwise in the spatial variable by
$(\hat{\Delta}u)(\cdot,s):=\Delta(u(\cdot,s))$ for almost every
$s\in[a,b]$. Similarly, define the lifted hard factorization
$(\hat{\Psi},\hat V)$ by
$(\hat{\Psi}x)(\cdot,s):=\Psi(x(\cdot,s))$ and
$(\hat Vy)(t,s):=Vy(t,s)$, where
$\hat{\Psi}:L_{\mbf{H},e}^n\to L_{\mbf{H},e}^{\tilde n}$ and
$\hat V:L_2^{\tilde n}[a,b]\to L_2^{\tilde n}[a,b]$. 

The problem is to show that
$\hat{\Delta}:L_{\mbf{H},e}\to L_{\mbf{H},e}$ is well-defined and that
the temporal hard IQC~\eqref{eqn:temporal-IQC} is preserved under this
spatial lifting. Specifically, we seek to show that
\begin{equation}\label{eqn:spatial-IQC}
    \ip{p_T\hat{\Psi}\bmat{u\\ \hat{\Delta}u}}{p_T\hat V\hat{\Psi}\bmat{u\\ \hat{\Delta}u}}_{L_{\mbf{H}}}\geq0
\end{equation}
for all $T>0$ and $u\in L_{\mbf{H},e}$.



\section{Positive Negative multipliers yield hard factorizations \cite{6915700}}
The lifting of finite dimensional multipliers can be nicely combined with the hard factorization theory developed by Seiler.
This hard factorization satisfies the assumptions of the duality theory thus can be directly applied. The main result of \cite{6915700} is the establishment of 
\begin{enumerate}
    \item $(\Psi, M)$ hard factorizations.
    \item $P=P^\top$ is a solution to the KYP lemma. (no enforcement of $P>0$)
\end{enumerate}
for the class of postive-negative multipliers. Before stating the theorem we need to define $J$-spectral factors.
\begin{definition}
    $(\Psi, M)$ is called a $J_{n,m}$-spectral factor of $\Pi = \Pi^\sim\in\R\mathbb{L}_\infty^{(n+m)\times(n+m)}$ if $\Pi = \Psi^\sim M \Psi$, $M - \bmat{I_n & 0 \\ 0 & - I_m}$, and
    $\Psi,\Psi^{-1}\in\mathbb{RH}_\infty^{(n+m)\times(n+m)}$
\end{definition}
Using $J$-spectral factorizations, Seiler \cite{6915700} derives the following theorem.
\begin{theorem}
    Let $\Pi=\Pi^\sim$ and partition as $\bmat{\Pi_{11} & \Pi^\sim_{21}\\ \Pi_{21} & \Pi_{22}}$ where $\Pi_{11}\in\mathbb{RL}_\infty^{n\times n}$ and $\Pi_{22}^{m\times m}$. If
    $\Pi_{11}(j \omega)>0$ and $\Pi_{22}(j\omega)<0$, i.e. $\Pi$ is strictly positive-negative(PN), $\forall\mathbb{R}\cup\{\infty\}$, then
    \begin{enumerate}
        \item $\Pi$ has a $J_{n,m}$-spectral factorization $(\Psi, M)$.
        \item The $J_{n,m}$-spectral factorization $(\Psi,M)$ is a hard factorization of $\Pi$.
        \item For $G\in\mathbb{RH}_\infty^{n\times m}$, let $(\hat{A},\hat{B},\hat{C},\hat{D})$ denote the state-space realization of $\Psi\bmat{G\\I}$. All solutions
              $P=P^\top$ to the KYP LMI satisfy $P\geq 0$.
    \end{enumerate}
\end{theorem}
