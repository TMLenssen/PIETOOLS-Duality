\begin{lemma}[Block inversion]\label{lem:blkinv}
    Let $\mcl{S}$ be a linear bounded operator in a Hilbert space be partitioned as $\mcl{S} = \bmat{\mcl{S}_{11} & \mcl{S}_{12} \\ \mcl{S}_{21} & \mcl{S}_{22}}$, and $\mcl{S}_{11}$, $\mcl{S}_{22}$, $\mcl{S}'_{11}$, and $\mcl{S}'_{22}$ are invertible. Then 
    \begin{equation*}
        \mcl{S}^{-1} = \bmat{\mcl{S}'_{11} & \mcl{S}'_{12} \\ \mcl{S}'_{21} & \mcl{S}'_{22}}
    \end{equation*}
    with $\mcl{S}'_{11} = (\mcl{S}_{11} - \mcl{S}_{12}\mcl{S}^{-1}_{22}\mcl{S}_{21})^{-1}$, $\mcl{S}'_{12} = -\mcl{S}_{11}^{-1}\mcl{S}_{12}\mcl{S}'_{22}$, $\mcl{S}'_{21} = -\mcl{S}^{-1}_{22}\mcl{S}_{21}\mcl{S}'_{11}$, and $\mcl{S}'_{22} = (\mcl{S}_{22} - \mcl{S}_{21}\mcl{S}^{-1}_{11}\mcl{S}_{12})^{-1}$
\end{lemma}
\begin{proof}
    \begin{align*}
        \mcl{S}\mcl{S}^{-1}&=\bmat{\mcl{S}_{11} & \mcl{S}_{12} \\ \mcl{S}_{21} & \mcl{S}_{22}}\bmat{\mcl{S}'_{11} & -\mcl{S}^{-1}_{11}\mcl{S}_{12}\mcl{S}'_{22} \\ -\mcl{S}^{-1}_{22}\mcl{S}_{21}\mcl{S}'_{11} & \mcl{S}'_{22}}
        \\&= \bmat{(\mcl{S}_{11} - \mcl{S}_{12}\mcl{S}^{-1}_{22}\mcl{S}_{21})\mcl{S}'_{11} & (\mcl{S}_{12}-\mcl{S}_{11}\mcl{S}^{-1}_{11}\mcl{S}_{12})\mcl{S}'_{22} \\ (\mcl{S}_{21} - \mcl{S}_{22}\mcl{S}^{-1}_{22}\mcl{S}_{21})\mcl{S}'_{11} & (\mcl{S}_{22}-\mcl{S}_{21}\mcl{S}^{-1}_{11}\mcl{S}_{12})\mcl{S}'_{22}} = I
    \end{align*}
    Similar calculation shows $\mcl{S}^{-1}\mcl{S} = I$.
\end{proof}

\begin{lemma}[Generalized Duality]\label{thm:duality}
Given linear bounded operators $G_\Delta, G_{\Delta w}, G_{z\Delta}, G_{zw}$, and a bounded, linear uncertainty $\Delta \in \mathbf{\Delta}$. Assume the interconnection $[G, \Delta]$ is well-posed. The following statements are equivalent,
\begin{enumerate}
\item For any $w\in L_2^{n_w}[0,\infty)$, $w_\Delta\in\mbf{L}_{e,[a,b]}^{\mbf{n}_{w_\Delta}}$, $z\in L_2^{n_z}[0,\infty)$, and $z_\Delta \in \mbf{L}_{e,[a,b]}^{\mbf{n}_{z_\Delta}}$ that satisfy,
        \begin{equation}\label{eqn:primal_system}
            \bmat{z_\Delta\\ z}=\bmat{G_\Delta & G_{\Delta w}\\ G_{z\Delta} & G_{zw}}\bmat{w_\Delta \\ w}
        \end{equation}
        \begin{center}
            with, $w_\Delta = \Delta z_\Delta$
        \end{center}
        we have that the operator norm $\|[G,\Delta]\| \leq \gamma$.
\item For any $\bar{w}\in L_2^{n_z}[0,\infty)$, $\bar{w}_\Delta\in\mbf{L}_{e,[a,b]}^{\mbf{n}_{z_\Delta}}$, $\bar{z}\in L_2^{n_w}[0,\infty)$, and $\bar{z}_\Delta \in \mbf{L}_{e,[a,b]}^{\mbf{n}_{w_\Delta}}$ that satisfy,
        \begin{equation}\label{eqn:dual_system}
            \bmat{\bar{z}_\Delta\\ \bar{z}}=\bmat{G_\Delta^* & G_{z\Delta}^*\\ G_{\Delta w}^* & G_{zw}^*}\bmat{\bar{w}_\Delta \\ \bar{w}}
        \end{equation}
        \begin{center}
            with, $\hat{w}_\Delta = \Delta^* \hat{z}_\Delta$
        \end{center}
        we have that the operator norm $\|[G^*,\Delta^*]\|\leq \gamma$
\end{enumerate}
\end{lemma}
\begin{proof}
Before proving the theorem we will introduce the push-through rule
\begin{sublemma}\label{lem:1} (Push-Through Rule)
Let \( G_\Delta \) and \( \Delta \) be bounded linear operators on a Hilbert space. If \( (I - G_\Delta \Delta) \) is invertible, then \( (I - \Delta G_\Delta) \) is invertible and
\[
(I - \Delta G_\Delta)^{-1} \Delta = \Delta (I - G_\Delta \Delta)^{-1}.
\]
\end{sublemma}

\begin{proof}
First, we show that \( (I - G_\Delta \Delta) \) is invertible by verifying that its inverse is \( \mathcal{X} = I + G_\Delta (I - \Delta G_\Delta)^{-1} \Delta \). Compute the product:
\begin{align*}
\mathcal{X} (I - G_\Delta \Delta) &= \left(I + G_\Delta (I - \Delta G_\Delta)^{-1} \Delta \right) (I - G_\Delta \Delta) \\
&= I - G_\Delta \Delta + G_\Delta (I - \Delta G_\Delta)^{-1} (\Delta - \Delta G_\Delta  \Delta) \\
&= I - G_\Delta \Delta + G_\Delta (I - \Delta G_\Delta)^{-1} (I - \Delta G_\Delta) \Delta \\
&= I - G_\Delta \Delta + G_\Delta \Delta = I.
\end{align*}
A similar calculation shows \( (I - G_\Delta \Delta) \mathcal{X} = I \). Thus, \( (I - G_\Delta \Delta) \) is invertible. To prove the identity, observe that \( \Delta (I - G_\Delta \Delta) = \Delta - \Delta G_\Delta \Delta = (I - G_\Delta \Delta) \Delta \). Multiplying by \( (I - G_\Delta \Delta)^{-1} \) on the left and \( (I - \Delta G_\Delta)^{-1} \) on the right yields
\[
(I - G_\Delta \Delta)^{-1} \Delta = \Delta (I - \Delta G_\Delta)^{-1}.
\]
\end{proof}
Substitute $w_\Delta(t) = (\Delta z_\Delta)(t)$ in \eqref{eqn:primal_system}, and $\bar{w}_\Delta(t) = (\Delta^* \bar{z}_\Delta)(t)$ in \eqref{eqn:dual_system} to find,
\begin{align*}
    [G,\Delta] &= G_{zw} + G_{z\Delta}\Delta(I-G_\Delta\Delta)^{-1}G_{\Delta w}\\
    [G^*, \Delta^*] &= G_{zw}^* + G_{\Delta w}^*\Delta^*(I-G_\Delta^*\Delta^*)^{-1}G_{z\Delta}^*
\end{align*}
\color{orange}Since the system is well-posed $(I-G_\Delta\Delta)^{-1}$ exist and is bounded, \color{black}$\|T\| = \|T^*\|$ for any bounded linear operator \cite[Thm 3.9]{kreyszig1991introductory} and by Sublemma~\ref{lem:1} we find
\begin{gather*}
    \|[G,\Delta]\| = \|[G,\Delta]^*\|= \|(G_{zw} + G_{z\Delta}\Delta(I-G_\Delta\Delta)^{-1}G_{\Delta w})^*\| \\= 
    \|(G_{zw} + G_{z\Delta}(I-\Delta G_\Delta)^{-1}\Delta G_{\Delta w})^*\|\\= \|(G_{zw}^* + G_{\Delta w}^*\Delta^*(I-G_\Delta^*\Delta^*)^{-1}G_{z\Delta}^*)\| = \|[G^*,\Delta^*]\|\leq\gamma
\end{gather*}
\end{proof}
