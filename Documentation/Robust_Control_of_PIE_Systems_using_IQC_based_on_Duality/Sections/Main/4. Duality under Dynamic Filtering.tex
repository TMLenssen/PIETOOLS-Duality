\begin{theorem}[Dual $L_2$-gain
{\cite[Thm.~9]{shivaknew}}]\label{thm:dual-l2-gain}
For a PIE system with corresponding PI operators $G\{\mcl{T}, \mcl{A}, \mcl{B},
\mcl{C}, \mcl{D}\}\in\Hinf$, the following statements are equivalent.
\begin{enumerate}
\item For any $w\in\mbf{L}_{e,[a,b]}^{\mbf{n}_w}$, and $\mbf{x}(0)=0$, $G$
satisfies $\|p_T {z}\| \leq \|p_T {w}\|$.
\item For any $\underline{w}\in\mbf{L}_{e,[a,b]}^{\mbf{n}_z}$, and
$\underline{\mbf{x}}(0)=0$, $G^\top$ satisfies $\|p_T \underline{z}\| \leq
\|p_T \underline{w}\|$.
\end{enumerate}
\end{theorem}


\begin{definition}[Primal and Dual Dynamic Filters]\label{def:dynamic-filters}
Let $\Theta=\bmat{\Theta_z & \Theta_{zw}\\ \Theta_{wz} & \Theta_w}$ satisfy
$\Theta,\Theta^{-1}\in\Hinf$. Let $\hat{\mcl{J}}:=\bmat{0&I\\ -I&0}$.
The dual filter associated with $\Theta$ is defined by the
dual operation
\begin{equation}\label{eq:dual-filter-operation}
\mbf{D}(\Theta)
:=
\left(\hat{\mcl{J}}^*\Theta^\top\hat{\mcl{J}}\right)^{-1}
=
\bmat{\Theta_w^\top & -\Theta_{zw}^\top\\
-\Theta_{wz}^\top & \Theta_z^\top}^{-1}.
\end{equation}
Note $\mbf{D}(\Theta)\in\Hinf$. Since $\hat{\mcl{J}}^*=\hat{\mcl{J}}^{-1}$,
$\hat{\mcl{J}}^*=-\hat{\mcl{J}}$, $(\Theta^\top)^\top=\Theta$, and
$(\Theta^{-1})^{-1}=\Theta$, this operation is an involution:
\begin{equation}\label{eq:dual-filter-involution}
\mbf{D}(\mbf{D}(\Theta))=\Theta.
\end{equation}
The filtered primal and dual systems are obtained
from
\begin{equation}\label{eq:filtered-primal-signals}
\Theta\bmat{z(t) \\ w(t)}
=
\bmat{\tilde{z}(t) \\ \tilde{w}(t)}, \qquad \mbf{D}(\Theta)
\bmat{\underline{z}(t) \\ \underline{w}(t)}
=
\bmat{\tilde{\underline{z}}(t) \\ \tilde{\underline{w}}(t)}.
\end{equation}
\end{definition}

The filter $\Theta$ admits the state-space realization
\begin{equation}\label{eq:primal-filter-realization}
\bmat{\mcl{T}_{\Theta}\dot{\mbf{x}}_{\Theta}(t)\\ \tilde{z}(t) \\ \tilde{w}(t)}
=
\bmat{\mcl{A}_{\Theta} & \mcl{B}_{\Theta}^z & \mcl{B}_{\Theta}^w \\
\mcl{C}_{\Theta}^z & \mcl{D}_{\Theta}^z & \mcl{D}_{\Theta}^{zw}\\
\mcl{C}_{\Theta}^w & \mcl{D}_{\Theta}^{wz} & \mcl{D}_{\Theta}^w}
\bmat{\mbf{x}_{\Theta}(t) \\ z(t) \\ w(t)},
\end{equation}
where the entries are PI operators of appropriate dimension.
Adopted from \cite{venkataraman}
\begin{definition}(Potapov--Ginzburg Transform)\label{def:potapov-ginzburg-transform}
Given $\Theta$ as in Definition~\ref{def:dynamic-filters}, the Potapov--Ginzburg transform denoted by $\Theta^\dagger$ is defined by
\begin{equation}\label{eq:potapov-ginzburg-transform}
\Theta^\dagger=\bmat{\Theta_{zw}\Theta_w^{-1} & \Theta_z-\Theta_{zw}\Theta_w^{-1}\Theta_{wz}\\ \Theta_w^{-1} & -\Theta_w^{-1}\Theta_{wz}},
\end{equation}
\end{definition}

\begin{lemma} (Potapov--Ginzburg Transform Equivalence)\label{lem:potapov-ginzburg-equivalence}
Let $\Theta, \mbf{D}(\Theta)\in \Hinf$ as in Definition~\ref{def:dynamic-filters}. Moreover, let $G$ and $G^\top$ be well-posed PIE systems such that
\begin{equation*}
\bmat{G\star\Theta^\dagger \\ I}\in\Hinf,
\qquad
\bmat{G^\top\star(\Theta^\dagger)^\top \\ I}\in\Hinf,
\end{equation*}
and $(\mcl{D}_{\Theta}^{wz} \mcl{D} + \mcl{D}_{\Theta}^w)^{-1}$ closed under inversion. Then, there invertible 
bounded $\Gamma\in\Pi_4$ on the primal side and $\Gamma_d\in\Pi_4$ on the dual side such that
\begin{itemize}
    \item $\Theta\bmat{G\\I} w = \bmat{G\star \Theta^\dagger \\ I}\Gamma w$ for all $w\in\mbf{L}_{e,[a,b]}^{\mbf{n}_w}$.
    \item $\mbf{D}(\Theta)\bmat{G^\top\\I} \underline{w} = \bmat{G^\top\star (\Theta^\dagger)^\top \\ I}\Gamma_d\underline{w}$ for all $\underline{w}\in\mbf{L}_{e,[a,b]}^{\mbf{n}_z}$.
\end{itemize}
\end{lemma}

See Appendix~\ref{proof:potapov-ginzburg-equivalence}.
\end{proof}
