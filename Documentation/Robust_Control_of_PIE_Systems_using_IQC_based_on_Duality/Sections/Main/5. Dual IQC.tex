


\begin{definition}(J-spectral factorization, Venkataraman et al. \cite{venkataraman})\label{def:factorization}
    ($\Theta, J_{\mbf{n}_z, \mbf{n}_w}$) is called a $J$-spectral factorization of the hard IQC ($\Psi, \mcl{M}_{\mcl{V}}$) if 
    \begin{equation}
        \begin{aligned}
        &\ip{p_T \Psi\bmat{z \\ w}}{p_T \mcl{M}_\mcl{V}\Psi\bmat{z \\ w}}_{\mbf L_{[a, b]}} \\&=
        \ip{p_T \Theta\bmat{z\\w}}{p_T J_{\mbf{n}_z, \mbf{n}_w}\Theta\bmat{z\\w}}_{\mbf L_{[a, b]}}
        \end{aligned}
    \end{equation}
    with $J_{\mbf{n}_z, \mbf{n}_w} = \bmat{I_{\mbf{n}_z} & 0 \\ 0 & -I_{\mbf{n}_w}}$, and $\Theta, \mbf{D}(\Theta)\in\Hinf$.
\end{definition}

\begin{theorem}[Primal--Dual IQC Feasibility]\label{thm:primal-dual-iqc-feasibility}
Let $P$ be a PIE realization of
Definition~\ref{def:uncertain-interconnection}, and let
$(\Theta,J_{\mbf{n}_z,\mbf{n}_w})$ be a $J$-spectral factorization of the
primal hard IQC. Let sufficiently small $\epsilon_p>0$ and $\epsilon_d>0$ denote the primal and
dual feasibility margins. Then the primal feasibility condition
\begin{equation}\label{eq:primal-iqc-feasibility}
\begin{gathered}
\begin{aligned}
\ip{p_T\Theta\bmat{z \\ w}}
{p_TJ_{\mbf{n}_z,\mbf{n}_w}\Theta\bmat{z \\ w}}_{\mbf L_{[a,b]}}
&\!\!\!\!\!\!\!\!\leq -\epsilon_p\|p_Tw\|^2_{\mbf L_{[a,b]}},
\end{aligned}\\
z=Pw,\qquad \forall w\in\mbf L_{e,[a,b]}^{\mbf{n}_w},\qquad T\geq0,
\end{gathered}
\end{equation}
holds if and only if the dual feasibility condition
\begin{equation}\label{eq:dual-iqc-feasibility}
\begin{gathered}
\begin{aligned}
\ip{p_T\mbf{D}(\Theta)\bmat{\underline{z} \\ \underline{w}}}
{p_TJ_{\mbf{n}_w,\mbf{n}_z}\mbf{D}(\Theta)\bmat{\underline{z} \\ \underline{w}}}_{\mbf L_{[a,b]}}
&\!\!\!\!\!\!\!\!\leq -\epsilon_d\|p_T\underline{w}\|^2_{\mbf L_{[a,b]}},
\end{aligned}\\
\underline{z}=P^\top\underline{w},\qquad
\forall \underline{w}\in\mbf L_{e,[a,b]}^{\mbf{n}_z},\qquad T\geq0,
\end{gathered}
\end{equation}
holds for $\epsilon_d > 0$.
\end{theorem}
\begin{proof}
See Appendix~\ref{proof:primal-dual-iqc-feasibility}.
\end{proof}

\begin{theorem}\label{thm:dual-KYP}
    Let the realization of $P$ satisfy the statement of Theorem~\ref{thm:robust-stability-performance}. 
    Let $\Delta \in \mbf{\Delta}$ and there exist \textbf{either}, a primal hard IQC factorization $\Psi_p$, and $\mcl{M}_{\mcl{V}_p}$, 
    with $\mcl{V}_p\in\Pi_4$, or a dual hard IQC factorization $\Psi_d$, and $\mcl{M}_{\mcl{V}_d}$, 
    with $\mcl{V}_d\in\Pi_4$, such that,
    \begin{itemize}
        \item The interconnection $P\star\Delta$ is well-posed for all $\Delta\in\mbf{\Delta}$.
        \item The IQC in Definition~\ref{def:uncertainty-iqc} is satisfied for all $\Delta\in\mbf{\Delta}$.
        \item The primal hard IQC $(\Psi_p,\mcl{M}_{\mcl{V}_p})$ has a $(\Theta,J_{\mbf{n}_z,\mbf{n}_w})$ factorization.
        \item The dual hard IQC $(\Psi_d,\mcl{M}_{\mcl{V}_d})$ has a $(\mbf{D}(\Theta),J_{\mbf{n}_w,\mbf{n}_z})$ factorization.
    \end{itemize}
    Suppose that \textbf{either} of the two statements hold for some $\mcl{Q}_p\succeq 0$ and $\mcl{P} = \mcl{P}^* \succeq 0 \in \Pi_4$. 
    \begin{itemize}
        \item \begin{minipage}[c]{\linewidth}
        \begin{equation}\label{eqn:primal-kyp-ineq}
            \bmat{\mcl{T}_p^*\mcl{P}\mcl{A}_p + (*) & \mcl{T}_p^*\mcl{P}\mcl{B}_p\\
            \mcl{B}_p^*\mcl{P}\mcl{T}_p & \epsilon_p I} + \bmat{\mcl{C}^*_p \\ \mcl{D}^*_p}\mcl{V}_p\bmat{\mcl{C}_p & \mcl{D}_p} \preceq 0
        \end{equation}
        \end{minipage}
        \item \begin{minipage}[c]{\linewidth}
        \begin{equation}\label{eqn:dual-kyp-ineq}
            \bmat{\mcl{T}_d^*\mcl{P}\mcl{A}_d + (*)& \mcl{T}_d^*\mcl{P}\mcl{B}_d\\
            \mcl{B}_d^*\mcl{P}\mcl{T}_d & \epsilon_d I} + \bmat{\mcl{C}^*_d \\ \mcl{D}^*_d}\mcl{V}_d\bmat{\mcl{C}_d & \mcl{D}_d} \preceq 0
        \end{equation}
        \end{minipage}
    \end{itemize}
    Then for any $w,\underline{w}\in\mbf{L}_{[a,b]}$, if $\tilde{z}$ or $\tilde{\underline{z}}$ satisfies either
    \begin{equation}\label{eqn:primal-kyp-sys}
        \Psi_p\bmat{z\\w} := \left\{ \bmat{\mcl{T}_p \dot{\mbf{x}}(t) \\ \tilde{z}(t)} = \bmat{\mcl{A}_p & \mcl{B}_p\\
        \mcl{C}_p & \mcl{D}_p}\bmat{\mbf{x} \\ w(t)}\right.
    \end{equation}
    or
    \begin{equation}\label{eqn:dual-kyp-sys}
        \Psi_d\bmat{\underline{z}\\\underline{w}} := \left\{ \bmat{\mcl{T}_d \dot{\underline{\mbf{x}}}(t) \\ \tilde{\underline{z}}(t)} = \bmat{\mcl{A}_d & \mcl{B}_d\\
        \mcl{C}_d & \mcl{D}_d}\bmat{\underline{\mbf{x}} \\ \underline{w}(t)}\right.
    \end{equation}
    Then the interconnection $P\star\Delta$ with $\mbf{x}(0) = 0$ is robustly stable and robust performance is achieved.
\end{theorem}
\begin{proof}
    See Appendix~\ref{proof:dual-KYP}
\end{proof}




J-spectral factorizations exist for many performance and uncertainty
multipliers.  Following \cite{veenman2016robust}, we collect the relevant
multiplier families as classes.

\subsection{Classes of J-Spectral Factorizations for Performance Multipliers}

\begin{class}[Induced $L_2$-gain]
The induced $L_2$-gain multiplier class is given by
\begin{equation}
\mcl{V}_p
=
\bmat{
\gamma^{-1}I_{\mbf{n}_{z_p}} & 0 \\
0 & -\gamma I_{\mbf{n}_{w_p}}
},
\qquad \gamma>0 .
\end{equation}
Robust performance then guarantees
$\sup_{\Delta\in\mbf{\Delta}}\|G\star\Delta\|_{\mbf L_{[a,b]}}\leq\gamma$.
A $J$-spectral factorization is
\begin{equation}
\mcl{V}_p
=
\Theta_\gamma^*
J_{\mbf{n}_{z_p},\mbf{n}_{w_p}}
\Theta_\gamma,
\qquad
\Theta_\gamma
=
\bmat{
\gamma^{-\frac{1}{2}}I_{\mbf{n}_{z_p}} & 0 \\
0 & \gamma^{\frac{1}{2}} I_{\mbf{n}_{w_p}}
}.
\end{equation}
The associated dual factor is
\begin{equation}
\mbf{D}(\Theta_\gamma)
=
\bmat{
\gamma^{-\frac{1}{2}}I_{\mbf{n}_{w_p}} & 0 \\
0 & \gamma^{\frac{1}{2}} I_{\mbf{n}_{z_p}}
},
\qquad
J_{\mbf{n}_{w_p},\mbf{n}_{z_p}} .
\end{equation}
\end{class}

\begin{class}[Passivity]
The passivity multiplier class is
\begin{equation}
\mcl{V}_p
=
\bmat{
0 & I_{\mbf{n}} \\
I_{\mbf{n}} & 0
}.
\end{equation}
Robust performance means that $G\star\Delta$ is strictly input passive for all
$\Delta\in\mbf{\Delta}$. A $J$-spectral factorization is
\begin{equation}
\mcl{V}_p
=
\Theta_{\rm pas}^*
J_{\mbf{n},\mbf{n}}
\Theta_{\rm pas},
\qquad
\Theta_{\rm pas}
=
\frac{1}{\sqrt{2}}
\bmat{
I_{\mbf{n}} & I_{\mbf{n}} \\
I_{\mbf{n}} & -I_{\mbf{n}}
}.
\end{equation}
The associated dual factor is
\begin{equation}
\mbf{D}(\Theta_{\rm pas})
=
\frac{1}{\sqrt{2}}
\bmat{
-I_{\mbf{n}} & -I_{\mbf{n}} \\
-I_{\mbf{n}} & I_{\mbf{n}}
},
\qquad
J_{\mbf{n},\mbf{n}} .
\end{equation}
\end{class}

\subsection{Classes of J-Spectral Factorizations for Uncertainty Multipliers}

Many finite-dimensional uncertainty IQCs have a strict positive-negative (PN)
multiplier structure.
\begin{definition}[Strict PN multiplier {\cite{6915700}}]
Let $\Pi=\Pi^\sim\in\mathbb{RL}_\infty^{(n_z+n_w)\times(n_z+n_w)}$, with
$\Pi_{11}\in\mathbb{RL}_\infty^{n_z\times n_z}$ and
$\Pi_{22}\in\mathbb{RL}_\infty^{n_w\times n_w}$. The multiplier $\Pi$ is
strict positive-negative if there exists $\epsilon>0$ such that
\begin{equation*}
\Pi_{11}(j\omega)\succeq\epsilon I,
\qquad
\Pi_{22}(j\omega)\preceq-\epsilon I,
\qquad
\forall\omega\in\R\cup\{\infty\}.
\end{equation*}
\end{definition}
Moreover, in this finite-dimensional setting, \cite{6915700} shows that if the
primal IQC is a strict PN multiplier, then a $J$-spectral factorization exists
according to Definition~\ref{def:factorization}. This factorization is hard,
and all solutions $P=P^\top$ to the associated KYP lemma satisfy $P\geq0$.
Also, \cite{venkataraman} shows that the primal IQC is a strict PN multiplier
if and only if the corresponding dual multiplier is a strict PN multiplier.
Classes 1, 2, 3, 5, 8, and 11 in \cite{veenman2016robust} are strict PN
multipliers. Classes 4, 6, 7, 12, 13, 14, and 15 are strict PN after imposing
the additional constraints stated in \cite{veenman2016robust}.

\begin{class}[Real repeated scalar parameters]
Consider the finite-dimensional uncertainty class
$\mbf{\Delta}_{\rm rr}:=\{\delta I_{n_q}:\delta\in\R,\ |\delta|\leq\alpha\}$.
Define the static matrix scaling
\begin{equation}
\begin{gathered}
\Sigma_\alpha
:=
\bmat{
\alpha^{-\frac{1}{2}}I_{n_q} & 0 \\
0 & \alpha^{\frac{1}{2}}I_{n_q}
}
\in\R^{2n_q\times 2n_q},\\
\Psi_{\rm rr}:=
\bmat{\psi & 0 \\ 0 & \psi}.
\end{gathered}
\end{equation}
With the convention $w_\Delta=\Delta z_\Delta$, the IQC holds with
\begin{equation}
\Pi_{\rm rr}
=
\Sigma_\alpha^*
\Psi_{\rm rr}^*
\bmat{\mcl{Q} & \mcl{S} \\ \mcl{S}^* & \mcl{R}}
\Psi_{\rm rr}
\Sigma_\alpha,
\end{equation}
for all $\Delta\in\mbf{\Delta}_{\rm rr}$ if
$\mcl{Q}=\mcl{Q}^*=-\mcl{R}\in\R^{n_q\times n_q}$,
$\mcl{S}\in\R^{n_q\times n_q}$, and
\begin{equation}
\psi^*\mcl{Q}\psi\prec0,
\qquad
\psi^*(\mcl{S}+\mcl{S}^*)\psi=0 .
\end{equation}
Here $\alpha$ is the radius of the real repeated uncertainty block and appears
only through the finite-dimensional matrix $\Sigma_\alpha$. Since the unscaled
dynamic multiplier is strict PN, there exists a hard $J$-spectral factorization
\begin{equation}
\Psi_{\rm rr}^*
\bmat{\mcl{Q} & \mcl{S} \\ \mcl{S}^* & \mcl{R}}
\Psi_{\rm rr}
=
\Theta_{\rm rr}^*
J_{n_q,n_q}
\Theta_{\rm rr},
\qquad
\Theta_{\rm rr},\mbf{D}(\Theta_{\rm rr})\in\Hinf .
\end{equation}
Thus $\alpha$ remains outside the $J$-spectral factorization:
\begin{equation}
\Pi_{\rm rr}
=
\Sigma_\alpha^*
\Theta_{\rm rr}^*
J_{n_q,n_q}
\Theta_{\rm rr}
\Sigma_\alpha.
\end{equation}
The corresponding unscaled dual multiplier is also strict PN. Hence there
exist dual-side parameters $\psi_d,\mcl{Q}_d,\mcl{S}_d,\mcl{R}_d$ such that
\begin{equation}
\begin{aligned}
\mbf{D}(\Theta_{\rm rr})^*
J_{n_q,n_q}
\mbf{D}(\Theta_{\rm rr})
&=
\bmat{\psi_d & 0 \\ 0 & \psi_d}^*
\bmat{\mcl{Q}_d & \mcl{S}_d \\ \mcl{S}_d^* & \mcl{R}_d} \\
&\qquad
\bmat{\psi_d & 0 \\ 0 & \psi_d},
\end{aligned}
\end{equation}
where $\mcl{Q}_d=\mcl{Q}_d^*=-\mcl{R}_d\in\R^{n_q\times n_q}$,
$\mcl{S}_d\in\R^{n_q\times n_q}$, and
\begin{equation}
\psi_d^*\mcl{Q}_d\psi_d\prec0,
\qquad
\psi_d^*(\mcl{S}_d+\mcl{S}_d^*)\psi_d=0 .
\end{equation}
The dual operation is multiplicative for compatible invertible factors:
\begin{equation}
\mbf{D}(\Psi_1\Psi_2)=\mbf{D}(\Psi_1)\mbf{D}(\Psi_2).
\end{equation}
Moreover, the static scaling is self-dual,
\begin{equation}
\mbf{D}(\Sigma_\alpha)
=
(\hat{\mcl{J}}^*\Sigma_\alpha^\top\hat{\mcl{J}})^{-1}
=
\Sigma_\alpha .
\end{equation}
Therefore the scaled dual factor is
\begin{equation}
\mbf{D}(\Theta_{\rm rr}\Sigma_\alpha)
=
\mbf{D}(\Theta_{\rm rr})\Sigma_\alpha,
\end{equation}
and the scaled dual-side multiplier is obtained by applying the same static
matrix scaling outside the dual $J$-spectral factorization:
\begin{equation}
\Pi_{{\rm rr},d}
=
\Sigma_\alpha^*
\mbf{D}(\Theta_{\rm rr})^*
J_{n_q,n_q}
\mbf{D}(\Theta_{\rm rr})
\Sigma_\alpha .
\end{equation}
\end{class}
