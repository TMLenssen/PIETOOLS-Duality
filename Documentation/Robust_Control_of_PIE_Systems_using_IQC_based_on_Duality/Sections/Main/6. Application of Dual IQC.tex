
Theorem~\ref{thm:dual-KYP} provides a robust-performance certificate for PIE
systems in terms of the dual IQC. For analysis, this merely reformulates the
primal certificate, since primal--dual equivalence preserves feasibility. The
principal benefit of the dual formulation therefore appears in controller
synthesis. Both controller constructions below use the same synthesis LPI,
obtained by applying \eqref{eqn:dual-kyp-ineq} to the corresponding closed-loop
PIE. The choice of dual uncertainty filter changes only the realization of the
controller recovered from this LPI. A related application of IQC duality to
convex feedforward controller synthesis is given by Venkataraman and
Seiler~\cite{venkataraman}.

\end{equation}

\begin{theorem}
    Let the realization of $P$ satisfy Definition~\ref{def:uncertain-interconnection}. Let $\Delta\in\mbf{\Delta}$, there exist dual hard IQC
    $(\underline{\Psi}, \mcl{M}_{\underline{\mcl{V}}})$, such that,
    \begin{itemize}
        \item The interconnection $P\star\Delta$ is well-posed for all $\Delta\in\mbf{\Delta}$.
        \item The IQC in Definition~\ref{def:uncertainty-iqc} is satisfied for all $\Delta\in\mbf{\Delta}$.
        \item The dual hard IQC $(\underline{\Psi},\mcl{M}_{\underline{\mcl V}})$ has a $(\mbf{D}(\Psi)\mcl{M}_{\mbf{D}(\mcl{U})},J_{\mbf{n}_w,\mbf{n}_z})$ factorization.
    \end{itemize}
     Moreover there exist, $\mcl{Z},\mcl{P}\in\Pi_4$ with $\mcl{P} = \mcl{P}^* \succ 0$, such that the following statement holds
    \begin{equation}\label{eqn:dual-controller-kyp-ineq}
        \begin{gathered}
            \bmat{
\underline{\mcl T}^*\mcl P\underline{\mcl A}_0
+
\underline{\mcl T}^*\mcl Z\bmat{\mcl B_u^*&0}
+(*)
&
\underline{\mcl T}^*\mcl P\underline{\mcl B}_0
+
\underline{\mcl T}^*\mcl Z\mcl D_{zu}^*
\\
(*)&
\underline\epsilon I
}
\\+
\bmat{\underline{\mcl C}^*\\\underline{\mcl D}^*}
\underline{\mcl V}
\bmat{\underline{\mcl C}&\underline{\mcl D}}
\preceq0.
        \end{gathered}
\end{equation}
wherewhere
\begin{equation*}
\bmat{\underline{\mcl A}_0&\underline{\mcl B}_0}
:=
\bmat{
\mcl A_P^*&0&\mcl C_P^*\\
\bmat{-\mcl C_\Psi^{w*}&\mcl C_\Psi^{z*}}
\underline{\mcl D}_\Psi^{-1}
\bmat{\mcl B_P^*\\0}
&
\mcl A_\Psi^*
-
\bmat{-\mcl C_\Psi^{w*}&\mcl C_\Psi^{z*}}
\underline{\mcl D}_\Psi^{-1}
\bmat{-\mcl B_\Psi^{w*}\\\mcl B_\Psi^{z*}}
&
\bmat{-\mcl C_\Psi^{w*}&\mcl C_\Psi^{z*}}
\underline{\mcl D}_\Psi^{-1}
\bmat{0\\I}
}.
\end{equation*}
The output rows are independent of the controller:
\begin{equation*}
\bmat{\underline{\mcl C}&\underline{\mcl D}}
:=
\bmat{
\underline{\mcl D}_\Psi^{-1}
\bmat{\mcl B_P^*\\0}
&
-
\underline{\mcl D}_\Psi^{-1}
\bmat{-\mcl B_\Psi^{w*}\\\mcl B_\Psi^{z*}}
&
\underline{\mcl D}_\Psi^{-1}
\bmat{0\\I}
}.
\end{equation*}
With
\[
\underline{\mcl T}:=\bmat{\mcl T_P^*&0\\0&\mcl T_\Psi^*},
\qquad
\mcl Z:=\mcl P\bmat{\mcl K_P^*\\\mcl K_\Psi^*},
\]
\end{theorem}


\subsection{\texorpdfstring{$\underline{\Psi}_\Delta = I$}
{Dual-filter identity} controller realization}

When $\Psi_\Delta=I$ and $\underline{\Psi}_\Delta=I$, no multiplier state is
required in the controller. For a state-feedback operator $\mcl{K}_P$, set
$u(t)=\mcl{K}_{P}\mbf{x}_P(t)$. The resulting closed-loop PIE
$\mcl{K}\star\mcl{P}$ is
\begin{equation}
\bmat{\partial_t(\mcl{T}_P\mbf{x}_P)(t)\\z(t)}
=
\bmat{
\mcl{A}_P+\mcl{B}_u\mcl{K}_P&\mcl{B}_P\\
\mcl{C}_P+\mcl D_{zu}\mcl{K}_P&\mcl{D}_P
}
\bmat{\mbf{x}_P(t)\\w(t)}.
\end{equation}

\subsection{Dynamic-filter controller realization}

Following \cite{7353149}, assume that the uncertainty $\Delta$ is known and
that $z_\Delta$, $w_\Delta=\Delta z_\Delta$, and $w_{\mathrm p}$ are available
to the controller. For $\underline{\Psi}_\Delta\neq I$, find the realization of $\Psi_\Delta = \mbf{D}(\underline{\Psi}_\Delta)$,
\begin{equation}
\bmat{
\mcl T_{{\Psi}_\Delta}
\dot{\mbf{x}}_{{\Psi}_\Delta}(t)\\
y(t)
}
=
\bmat{
\mcl A_{{\Psi}_\Delta}
&\mcl B_{{\Psi}_\Delta}^{z}
&\mcl B_{{\Psi}_\Delta}^{w}\\
I&0&0
}
\bmat{
\mbf{x}_{{\Psi}_\Delta}(t)\\
z_\Delta(t)\\
w_\Delta(t)
}.
\end{equation}
We impose the zero initial condition
$\mbf{x}_{{\Psi}_\Delta}(0)=0$. The second block row gives
$y(t)=\mbf{x}_{{\Psi}_\Delta}(t)$. The complete controller has the PIE
realization
\begin{equation}
\begin{aligned}
\bmat{
\mcl T_{{\Psi}_\Delta}
\dot{\mbf{x}}_{{\Psi}_\Delta}(t)\\
u(t)
}
&=
\bmat{
\mcl A_{{\Psi}_\Delta}
&0
&\mcl B_{{\Psi}_\Delta}^{z}
&\mcl B_{{\Psi}_\Delta}^{w}\\
\mcl K_\Psi&\mcl K_P & 0 & 0
}
\bmat{
\mbf{x}_{{\Psi}_\Delta}(t)\\
\mbf{x}(t)\\
z_\Delta(t)\\
w_\Delta(t)
}.
\end{aligned}
\end{equation}
Hence $\mbf{x}_{{\Psi}_\Delta}$ is the controller state,
$\mbf{x}$, $z_\Delta$, and $w_\Delta$ are its external inputs, and $u$ is its
output.
