\subsection{Proof of Theorem~\ref{thm:robust-stability-performance}}\label{proof:robust-stability-performance}
Suppose that \eqref{eq:robust-condition} holds. This implies,
\begin{equation*}
\ip{p_T \Psi_\Delta\bmat{P_\Delta \\ I}w_\Delta}{p_T\mcl{M}_{\mcl{V}_\Delta}\Psi_\Delta\bmat{P_\Delta \\ I}w_\Delta}_{\mbf L_{[a, b]}}
+\ip{p_T \Psi_{\mathrm{p}}\bmat{P_{\mathrm{p}\Delta} \\ 0}w_\Delta}{p_T\mcl{M}_{\mcl V_{\mathrm{p}}}\Psi_{\mathrm{p}}\bmat{P_{\mathrm{p}\Delta} \\ 0}w_\Delta}_{\mbf L_{[a, b]}}
\leq -\epsilon \|p_Tw_\Delta\|^2_{\mbf L_{[a, b]}}.
\end{equation*}
By Definition~\ref{def:robust-performance} we can then infer,
\begin{equation*}
\ip{p_T \Psi_\Delta\bmat{P_\Delta w_\Delta \\ w_\Delta}}{p_T\mcl{M}_{\mcl{V}_\Delta}\Psi_\Delta\bmat{P_\Delta w_\Delta \\ w_\Delta}}_{\mbf L_{[a, b]}}
\leq -\epsilon \|p_Tw_\Delta\|^2_{\mbf L_{[a, b]}}.
\end{equation*}
Then, using Theorem 6 in \cite{10156335}, we find that \textbf{robust stability}
is guaranteed.
Moreover, rewriting \eqref{eq:robust-condition} as,
\begin{equation*}
\ip{p_T\Psi_\Delta\bmat{z_\Delta \\ w_\Delta}}{p_T\mcl{M}_{\mcl{V}_\Delta}\Psi_\Delta\bmat{z_\Delta \\ w_\Delta}}_{\mbf L_{[a, b]}}
+\ip{p_T\Psi_{\mathrm{p}}\bmat{z_{\mathrm{p}} \\ w_{\mathrm{p}}}}{p_T\mcl{M}_{\mcl V_{\mathrm{p}}}\Psi_{\mathrm{p}}\bmat{z_{\mathrm{p}} \\ w_{\mathrm{p}}}}_{\mbf L_{[a, b]}}
\leq -\epsilon\bigl(\|p_T{w_\Delta}\|^2_{\mbf L_{[a, b]}}
+\|p_T w_{\mathrm{p}}\|^2_{\mbf L_{[a, b]}}\bigr),
\end{equation*}
and given that $\Psi_\Delta$ and $\mcl{M}_{\mcl{V}_\Delta}$ satisfy the IQC in
Definition~\ref{def:uncertainty-iqc} leads to,
\begin{equation*}
\ip{p_T\Psi_{\mathrm{p}}\bmat{z_{\mathrm{p}} \\ w_{\mathrm{p}}}}{p_T\mcl{M}_{\mcl V_{\mathrm{p}}}\Psi_{\mathrm{p}}\bmat{z_{\mathrm{p}} \\ w_{\mathrm{p}}}}_{\mbf L_{[a, b]}}
\leq -\epsilon \|p_T w_{\mathrm{p}}\|^2_{\mbf L_{[a,b]}}.
\end{equation*}
Thus by Definition~\ref{def:robust-performance} \textbf{robust performance} is achieved.
