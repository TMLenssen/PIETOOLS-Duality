\subsection{Proof of Theorem~\ref{thm:robust-stability-performance}}\label{proof:robust-stability-performance}
Suppose that \eqref{eq:robust-condition} holds. This implies,
\begin{equation*}
\ip{p_T \Psi_\Delta\bmat{P_\Delta \\ I}w_\Delta}{p_T\mcl{M}_{\mcl{V}_\Delta}\Psi_\Delta\bmat{P_\Delta \\ I}w_\Delta}_{L_{[a,b]}}
+\ip{p_T \Psi_{\mrm{p}}\bmat{P_{\mrm{p}\Delta} \\ 0}w_\Delta}{p_T\mcl{M}_{\mcl V_{\mrm{p}}}\Psi_{\mrm{p}}\bmat{P_{\mrm{p}\Delta} \\ 0}w_\Delta}_{L_{[a,b]}}
\leq -\epsilon \|p_Tw_\Delta\|^2_{L_{[a,b]}}.
\end{equation*}
By Definition~\ref{def:robust-performance} we can then infer,
\begin{equation*}
\ip{p_T \Psi_\Delta\bmat{P_\Delta w_\Delta \\ w_\Delta}}{p_T\mcl{M}_{\mcl{V}_\Delta}\Psi_\Delta\bmat{P_\Delta w_\Delta \\ w_\Delta}}_{L_{[a,b]}}
\leq -\epsilon \|p_Tw_\Delta\|^2_{L_{[a,b]}}.
\end{equation*}
Then, using Theorem 6 in \cite{10156335}, we find that \textbf{robust stability}
is guaranteed.
Moreover, rewriting \eqref{eq:robust-condition} as,
\begin{equation*}
\ip{p_T\Psi_\Delta\bmat{z_\Delta \\ w_\Delta}}{p_T\mcl{M}_{\mcl{V}_\Delta}\Psi_\Delta\bmat{z_\Delta \\ w_\Delta}}_{L_{[a,b]}}
+\ip{p_T\Psi_{\mrm{p}}\bmat{z_{\mrm{p}} \\ w_{\mrm{p}}}}{p_T\mcl{M}_{\mcl V_{\mrm{p}}}\Psi_{\mrm{p}}\bmat{z_{\mrm{p}} \\ w_{\mrm{p}}}}_{L_{[a,b]}}
\leq -\epsilon\bigl(\|p_T{w_\Delta}\|^2_{L_{[a,b]}}
+\|p_T w_{\mrm{p}}\|^2_{L_{[a,b]}}\bigr),
\end{equation*}
and given that $\Psi_\Delta$ and $\mcl{M}_{\mcl{V}_\Delta}$ satisfy the IQC in
Definition~\ref{def:uncertainty-iqc} leads to,
\begin{equation*}
\ip{p_T\Psi_{\mrm{p}}\bmat{z_{\mrm{p}} \\ w_{\mrm{p}}}}{p_T\mcl{M}_{\mcl V_{\mrm{p}}}\Psi_{\mrm{p}}\bmat{z_{\mrm{p}} \\ w_{\mrm{p}}}}_{L_{[a,b]}}
\leq -\epsilon \|p_T w_{\mrm{p}}\|^2_{L_{[a,b]}}.
\end{equation*}
Thus by Definition~\ref{def:robust-performance} \textbf{robust performance} is achieved.
