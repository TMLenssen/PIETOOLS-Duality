\subsection{Proof of Lemma~\ref{lem:mixed-IQC}}\label{proof:mixed-IQC}
To show sufficieny, assume~\eqref{eqn:primal-mixed} is satisfied. We need to show that~\eqref{eqn:dual-mixed} is also satisfied.
Factorize~\eqref{eqn:primal-mixed} as 
    \begin{equation}\label{eqn:expand}
        \ip{p_T\bmat{P\\I}w}{p_T\bmat{\gamma_1 I \\ \vdots \\ \gamma_n I}^* \bmat{\Theta_1^*J\Theta_1 & \cdots & 0\\
        \vdots & \ddots & \vdots\\
        0 & \cdots & \Theta_n^*J\Theta_n}\bmat{\gamma_1 I \\ \vdots \\ \gamma_n I}\bmat{P\\I}w} \leq -\epsilon\|p_T w\|^2_{\mbf{L}_{[a,b]}}
    \end{equation}
    Define $\hat{z} = \bmat{\gamma_1 I & \cdots & \gamma_n I }^* z$, $\hat{w} = \bmat{\gamma_1 I & \cdots & \gamma_n I }^* w$, then
    $\bmat{\frac{1}{\gamma_1}I & \cdots & \frac{1}{\gamma_n}I}\hat{w} = w$. Then we can reorganize~\eqref{eqn:expand} to show
    \begin{equation*}
        \ip{p_T\Theta\bmat{\hat{P}\\I}\hat{w}}{p_T J\Theta\bmat{\hat{P}\\I}\hat{w}} \leq -\epsilon\|p_T \bmat{\frac{1}{\gamma_1}I & \cdots & \frac{1}{\gamma_n}}\hat{w}\|^2_{\mbf{L}_{[a,b]}}
    \end{equation*}
    where,
    \begin{equation*}
        \hat{P}\hat{w} := \left\{\bmat{\mcl{T}\dot{\mbf{x}}(t)\\\hat{z}} = \bmat{\mcl{A} & \mcl{B} \bmat{\frac{1}{\gamma_1}I & \cdots & \frac{1}{\gamma_n}}\\
        \bmat{\gamma_1 I & \cdots & \gamma_n I }^*\mcl{C} & \bmat{\gamma_1 I & \cdots & \gamma_n I }^*\mcl{D} \bmat{\frac{1}{\gamma_1}I & \cdots & \frac{1}{\gamma_n}}}\bmat{\mbf{x}(t)\\\hat{w}(t)} \right.
    \end{equation*}
    then,
    \begin{equation*}
        \hat{P}^\top\underline{\hat{w}} := \left\{\bmat{\mcl{T}\dot{\underline{\mbf{x}}}(t)\\\underline{\hat{z}}} = \bmat{\mcl{A}^* &  \mcl{C}^*\bmat{\gamma_1 I & \cdots & \gamma_n I } \\
        \bmat{\frac{1}{\gamma_1}I & \cdots & \frac{1}{\gamma_n}}^*\mcl{B}^*& \bmat{\frac{1}{\gamma_1}I & \cdots & \frac{1}{\gamma_n}}^*\mcl{D}^*\bmat{\gamma_1 I & \cdots & \gamma_n I } }\bmat{\underline{\mbf{x}}(t)\\\underline{\hat{w}}(t)} \right.
    \end{equation*}
    and by Theorem~\ref{thm:primal-dual-iqc-feasibility}
    \begin{equation*}
        \ip{p_T\mbf{D}(\Theta)\bmat{\hat{P}^\top\\I}\underline{\hat{w}}}{p_T J\mbf{D}(\Theta)\bmat{\hat{P}^\top\\I}\underline{\hat{w}}} \leq -\underline{\epsilon}\|p_T \bmat{\frac{1}{\gamma_1}I & \cdots & \frac{1}{\gamma_n}}\underline{\hat{w}}\|^2_{\mbf{L}_{[a,b]}}
    \end{equation*}
