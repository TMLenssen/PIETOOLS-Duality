\subsection{Proof of Lemma~\ref{lem:potapov-ginzburg-equivalence}}\label{proof:potapov-ginzburg-equivalence}

We first show the primal derivation explicitly. To show
sufficiency, first augment the primal system with the
Potapov--Ginzburg transform of the primal filter, i.e. $P\star\Theta^\dagger$.
$\Theta^\dagger$ admits the state-space representation
\begin{equation}\label{eq:potapov-ginzburg-realization}
\bmat{
\mcl{T}_{\Theta}\dot{\mbf{x}}_{\Theta}(t)\\
\tilde{z}(t)\\
w(t)}
=
\bmat{
\mcl{A}_{\Theta}^\dagger
    & \mcl{B}_{\Theta}^{\dagger,\tilde{w}}
    & \mcl{B}_{\Theta}^{\dagger,z}\\
\mcl{C}_{\Theta}^{\dagger,z}
    & \mcl{D}_{\Theta}^{\dagger,z\tilde{w}}
    & \mcl{D}_{\Theta}^{\dagger,z}\\
\mcl{C}_{\Theta}^{\dagger,w}
    & \mcl{D}_{\Theta}^{\dagger,w\tilde{w}}
    & \mcl{D}_{\Theta}^{\dagger,\mathrm{wz}}}
\bmat{\mbf{x}_{\Theta}(t)\\ \tilde{w}(t)\\ z(t)}.
\end{equation}
With $W_{\Theta}:=(\mcl{D}_{\Theta}^w)^{-1}$, the transformed blocks are
\begin{align*}
\mcl{A}_{\Theta}^\dagger
&:= \mcl{A}_{\Theta}-\mcl{B}_{\Theta}^w W_{\Theta}\mcl{C}_{\Theta}^w,&
\mcl{B}_{\Theta}^{\dagger,\tilde{w}}
&:= \mcl{B}_{\Theta}^w W_{\Theta},&
\mcl{B}_{\Theta}^{\dagger,z}
&:= \mcl{B}_{\Theta}^z-\mcl{B}_{\Theta}^w W_{\Theta}\mcl{D}_{\Theta}^{wz},\\
\mcl{C}_{\Theta}^{\dagger,z}
&:= \mcl{C}_{\Theta}^z-\mcl{D}_{\Theta}^{zw}W_{\Theta}\mcl{C}_{\Theta}^w,&
\mcl{D}_{\Theta}^{\dagger,z}
&:= \mcl{D}_{\Theta}^z-\mcl{D}_{\Theta}^{zw}W_{\Theta}\mcl{D}_{\Theta}^{wz},&
\mcl{D}_{\Theta}^{\dagger,z\tilde{w}}
&:= \mcl{D}_{\Theta}^{zw}W_{\Theta},\\
\mcl{C}_{\Theta}^{\dagger,w}
&:= -W_{\Theta}\mcl{C}_{\Theta}^w,&
\mcl{D}_{\Theta}^{\dagger,w\tilde{w}}
&:= W_{\Theta},&
\mcl{D}_{\Theta}^{\dagger,wz}
&:= -W_{\Theta}\mcl{D}_{\Theta}^{wz}.
\end{align*}
Substitution of $z=\mcl{C}\mbf{x}+\mcl{D}w$ into
\eqref{eq:potapov-ginzburg-realization} and solving for $w$ gives the algebraic loop
\begin{equation}\label{eqn:solve_w}
w=
\Lambda_w\mcl{D}_{\Theta}^{\dagger,wz}\mcl{C}\mbf{x}
+\Lambda_w\mcl{C}_{\Theta}^{\dagger,w}\mbf{x}_{\Theta}
+\Lambda_w\mcl{D}_{\Theta}^{\dagger,w\tilde{w}}\tilde{w}.
\end{equation}
Define the algebraic loops
\begin{equation*}
\Lambda_w := (I-\mcl{D}_{\Theta}^{\dagger,wz}\mcl{D})^{-1},
\qquad
\Lambda_z := (I-\mcl{D}\mcl{D}_{\Theta}^{\dagger,wz})^{-1}.
\end{equation*}
Since $(\mcl{D}_{\Theta}^{wz} \mcl{D} + \mcl{D}_{\Theta}^w)^{-1} = (\mcl{D}_{\Theta}^w)^{-1}(I+(\mcl{D}_{\Theta}^w)^{-1}\mcl{D}_{\Theta}^{wz}\mcl{D})^{-1}
= (\mcl{D}_{\Theta}^w)^{-1}(I - \mcl{D}_{\Theta}^{\dagger,wz}\mcl{D})^{-1} $ invertible, $\Lambda_w, \Lambda_z$ invertible.
Substituting \eqref{eqn:solve_w} into \eqref{eqn:ss_primal} gives
\begin{align*}
\mcl{T}\dot{\mbf{x}}
&= (\mcl{A}
+\mcl{B}\Lambda_w\mcl{D}_{\Theta}^{\dagger,wz}\mcl{C})\mbf{x}
+\mcl{B}\Lambda_w\mcl{C}_{\Theta}^{\dagger,w}\mbf{x}_{\Theta}
+\mcl{B}\Lambda_w\mcl{D}_{\Theta}^{\dagger,w\tilde{w}}\tilde{w}.
\end{align*}
Using $\mcl{D}\Lambda_w=\Lambda_z\mcl{D}$ and
$I+\mcl{D}\Lambda_w\mcl{D}_{\Theta}^{\dagger,wz}=\Lambda_z$, we obtain
\begin{equation}\label{eqn:solve_z}
z
=
\Lambda_z\mcl{C}\mbf{x}
+\Lambda_z\mcl{D}\mcl{C}_{\Theta}^{\dagger,w}\mbf{x}_{\Theta}
+\Lambda_z\mcl{D}\mcl{D}_{\Theta}^{\dagger,w\tilde{w}}\tilde{w}.
\end{equation}
Substitution into the equations for $\dot{\mbf{x}}_{\Theta}$ and
$\tilde{z}$ yields $P\star\Theta^\dagger$
\begin{equation}\label{eq:primal-potapov-augmentation}
\bmat{\mcl{T}\dot{\mbf{x}}(t) \\ \mcl{T}_{\Theta}\dot{\mbf{x}}_{\Theta}(t)\\
\tilde{z}(t)} =
\bmat{\mcl{A} + \mcl{B}\Lambda_w\mcl{D}_{\Theta}^{\dagger,wz}\mcl{C} &
\mcl{B}\Lambda_w\mcl{C}_{\Theta}^{\dagger,w} &
\mcl{B}\Lambda_w\mcl{D}_{\Theta}^{\dagger,w\tilde{w}} \\
\mcl{B}_{\Theta}^{\dagger,z}\Lambda_z\mcl{C} & \mcl{A}_{\Theta}^\dagger +
\mcl{B}_{\Theta}^{\dagger,z}\Lambda_z\mcl{D}\mcl{C}_{\Theta}^{\dagger,w} &
\mcl{B}_{\Theta}^{\dagger,\tilde{w}} +
\mcl{B}_{\Theta}^{\dagger,z}\Lambda_z\mcl{D}\mcl{D}_{\Theta}^{\dagger,w\tilde{w}} \\
\mcl{D}_{\Theta}^{\dagger,z}\Lambda_z\mcl{C} & \mcl{C}_{\Theta}^{\dagger,z} +
\mcl{D}_{\Theta}^{\dagger,z}\Lambda_z\mcl{D}\mcl{C}_{\Theta}^{\dagger,w} &
\mcl{D}_{\Theta}^{\dagger,z\tilde{w}} +
\mcl{D}_{\Theta}^{\dagger,z}\Lambda_z\mcl{D}
\mcl{D}_{\Theta}^{\dagger,w\tilde{w}}}\bmat{\mbf{x}(t)
\\ \mbf{x}_{\Theta}(t) \\ \tilde{w}(t)}
\end{equation}
Define also
\begin{equation*}
\Lambda_c := (\mcl{D}_{\Theta}^{wz} \mcl{D} + \mcl{D}_{\Theta}^w)^{-1}.
\end{equation*}
The invertible congruence transformations, $\Gamma$ and $\Gamma^{-1}$, respectively, to recover \eqref{eq:scaled-primal-system-realization} and 
\eqref{eq:primal-potapov-augmentation}, respectively, are,
\begin{equation}\label{eq:pg-congruence-transform}
\bmat{\mbf{x}(t) \\ \mbf{x}_{\Theta}(t) \\ \tilde{w}(t)} = 
\bmat{I & 0 & 0\\ 
0 & I & 0\\
\mcl{D}_{\Theta}^{wz}\mcl{C} & \mcl{C}_{\Theta}^w & \Lambda_c^{-1}
}\bmat{\mbf{x}(t) \\ \mbf{x}_{\Theta}(t) \\ {w}(t)}
\qquad
\bmat{\mbf{x}(t) \\ \mbf{x}_{\Theta}(t) \\ w(t)} = 
\bmat{I & 0 & 0\\ 
0 & I & 0\\
-\Lambda_c\mcl{D}_{\Theta}^{wz} \mcl{C} & -\Lambda_c\mcl{C}_{\Theta}^w & \Lambda_c
}\bmat{\mbf{x}(t) \\ \mbf{x}_{\Theta}(t) \\ \tilde{w}(t)}
\end{equation}
The third row of \eqref{eq:pg-congruence-transform} is
\begin{equation}\label{eq:primal-pg-recovered-input-row}
\tilde{w}
=\mcl{D}_{\Theta}^{wz}\mcl{C}\mbf{x}
+\mcl{C}_{\Theta}^w\mbf{x}_{\Theta}
+(\mcl{D}_{\Theta}^{wz}\mcl{D}+\mcl{D}_{\Theta}^w)w.
\end{equation}
Using $z=\mcl{C}\mbf{x}+\mcl{D}w$, this is equivalently
\begin{equation*}
\tilde{w}
=\mcl{C}_{\Theta}^w\mbf{x}_{\Theta}
+\mcl{D}_{\Theta}^{wz}z
+\mcl{D}_{\Theta}^w w,
\end{equation*}
which is the scaled input row of $\Theta\bmat{P\\I}w$. We now apply \eqref{eq:pg-congruence-transform} directly to
\eqref{eq:primal-potapov-augmentation}.  The plant row of
\eqref{eq:primal-potapov-augmentation} gives
\begin{align*}
\mcl{T}\dot{\mbf{x}}
&=(\mcl{A}+\mcl{B}\Lambda_w\mcl{D}_{\Theta}^{\dagger,wz}\mcl{C})\mbf{x}
+\mcl{B}\Lambda_w\mcl{C}_{\Theta}^{\dagger,w}\mbf{x}_{\Theta}
+\mcl{B}\Lambda_w\mcl{D}_{\Theta}^{\dagger,w\tilde{w}}\tilde{w}\\
&=\mcl{A}\mbf{x}
+\mcl{B}(
\Lambda_w\mcl{D}_{\Theta}^{\dagger,wz}\mcl{C}\mbf{x}
+\Lambda_w\mcl{C}_{\Theta}^{\dagger,w}\mbf{x}_{\Theta}
+\Lambda_w\mcl{D}_{\Theta}^{\dagger,w\tilde{w}}\tilde{w})\\
&=\mcl{A}\mbf{x}+\mcl{B}w,
\end{align*}
where the last equality uses \eqref{eqn:solve_w}. The filter state
row of \eqref{eq:primal-potapov-augmentation} gives
\begin{align*}
\mcl{T}_{\Theta}\dot{\mbf{x}}_{\Theta}
&=\mcl{B}_{\Theta}^{\dagger,z}\Lambda_z\mcl{C}\mbf{x}
+(\mcl{A}_{\Theta}^\dagger+\mcl{B}_{\Theta}^{\dagger,z}
\Lambda_z\mcl{D}\mcl{C}_{\Theta}^{\dagger,w})\mbf{x}_{\Theta}\\
&\qquad
+(\mcl{B}_{\Theta}^{\dagger,\tilde{w}}+\mcl{B}_{\Theta}^{\dagger,z}
\Lambda_z\mcl{D}\mcl{D}_{\Theta}^{\dagger,w\tilde{w}})\tilde{w}\\
&=\mcl{A}_{\Theta}^\dagger\mbf{x}_{\Theta}
+\mcl{B}_{\Theta}^{\dagger,\tilde{w}}\tilde{w}
+\mcl{B}_{\Theta}^{\dagger,z}
(\Lambda_z\mcl{C}\mbf{x}
+\Lambda_z\mcl{D}\mcl{C}_{\Theta}^{\dagger,w}\mbf{x}_{\Theta}
+\Lambda_z\mcl{D}\mcl{D}_{\Theta}^{\dagger,w\tilde{w}}\tilde{w})\\
&=\mcl{A}_{\Theta}^\dagger\mbf{x}_{\Theta}
+\mcl{B}_{\Theta}^{\dagger,\tilde{w}}\tilde{w}
+\mcl{B}_{\Theta}^{\dagger,z} z\\
&=(\mcl{A}_{\Theta}-\mcl{B}_{\Theta}^wW_{\Theta}\mcl{C}_{\Theta}^w)\mbf{x}_{\Theta}
+\mcl{B}_{\Theta}^wW_{\Theta}
(\mcl{C}_{\Theta}^w\mbf{x}_{\Theta}+\mcl{D}_{\Theta}^{wz}z+\mcl{D}_{\Theta}^ww)\\
&\qquad
+(\mcl{B}_{\Theta}^z-\mcl{B}_{\Theta}^wW_{\Theta}\mcl{D}_{\Theta}^{wz})z\\
&=\mcl{A}_{\Theta}\mbf{x}_{\Theta}+\mcl{B}_{\Theta}^z z+\mcl{B}_{\Theta}^w w.
\end{align*}
The output row of \eqref{eq:primal-potapov-augmentation}, $W_{\Theta} = (\mcl{D}_{\Theta}^w)^{-1}$ and using \eqref{eqn:solve_z} gives
\begin{align*}
\tilde{z}
&=\mcl{D}_{\Theta}^{\dagger,z}\Lambda_z\mcl{C}\mbf{x}
+(\mcl{C}_{\Theta}^{\dagger,z}+\mcl{D}_{\Theta}^{\dagger,z}
\Lambda_z\mcl{D}\mcl{C}_{\Theta}^{\dagger,w})\mbf{x}_{\Theta}\\
&\qquad
+(\mcl{D}_{\Theta}^{\dagger,z\tilde{w}}+\mcl{D}_{\Theta}^{\dagger,z}
\Lambda_z\mcl{D}\mcl{D}_{\Theta}^{\dagger,w\tilde{w}})\tilde{w}\\
&=\mcl{C}_{\Theta}^{\dagger,z}\mbf{x}_{\Theta}
+\mcl{D}_{\Theta}^{\dagger,z\tilde{w}}\tilde{w}
+\mcl{D}_{\Theta}^{\dagger,z}
(\Lambda_z\mcl{C}\mbf{x}
+\Lambda_z\mcl{D}\mcl{C}_{\Theta}^{\dagger,w}\mbf{x}_{\Theta}
+\Lambda_z\mcl{D}\mcl{D}_{\Theta}^{\dagger,w\tilde{w}}\tilde{w})\\
&=\mcl{C}_{\Theta}^{\dagger,z}\mbf{x}_{\Theta}
+\mcl{D}_{\Theta}^{\dagger,z\tilde{w}}\tilde{w}
+\mcl{D}_{\Theta}^{\dagger,z} z\\
&=(\mcl{C}_{\Theta}^z-\mcl{D}_{\Theta}^{zw}W_{\Theta}\mcl{C}_{\Theta}^w)\mbf{x}_{\Theta}
+\mcl{D}_{\Theta}^{zw}W_{\Theta}
(\mcl{C}_{\Theta}^w\mbf{x}_{\Theta}+\mcl{D}_{\Theta}^{wz}z+\mcl{D}_{\Theta}^ww)\\
&\qquad
+(\mcl{D}_{\Theta}^z-\mcl{D}_{\Theta}^{zw}W_{\Theta}\mcl{D}_{\Theta}^{wz})z\\
&=\mcl{C}_{\Theta}^z\mbf{x}_{\Theta}+\mcl{D}_{\Theta}^z z+\mcl{D}_{\Theta}^{zw}w.
\end{align*}
Substituting $z=\mcl{C}\mbf{x}+\mcl{D}w$ in these recovered rows gives
\begin{equation}\label{eq:scaled-primal-system-realization}
\bmat{\mcl{T}\dot{\mbf{x}}(t)\\
\mcl{T}_{\Theta}\dot{\mbf{x}}_{\Theta}(t)\\
\tilde{z}(t)\\
\tilde{w}(t)}
=
\bmat{
\mcl{A} & 0 & \mcl{B}\\
\mcl{B}_{\Theta}^z\mcl{C} & \mcl{A}_{\Theta} &
\mcl{B}_{\Theta}^z\mcl{D}+\mcl{B}_{\Theta}^w\\
\mcl{D}_{\Theta}^z\mcl{C} & \mcl{C}_{\Theta}^z &
\mcl{D}_{\Theta}^z\mcl{D}+\mcl{D}_{\Theta}^{zw}\\
\mcl{D}_{\Theta}^{wz}\mcl{C} & \mcl{C}_{\Theta}^w &
\mcl{D}_{\Theta}^{wz}\mcl{D}+\mcl{D}_{\Theta}^w}
\bmat{\mbf{x}(t)\\ \mbf{x}_{\Theta}(t)\\ w(t)}.
\end{equation}
The realization in \eqref{eq:scaled-primal-system-realization} is the
direct augmentation $\Theta\bmat{P \\ I}w$. For necessity, applying the inverse
of \eqref{eq:pg-congruence-transform} to
\eqref{eq:scaled-primal-system-realization} recovers
\eqref{eq:primal-potapov-augmentation}. Thus we find $\Theta\bmat{P\\ I}{w} = \bmat{P \star \Theta^\dagger \\ I}\Gamma{w}$
for all ${w}\in\mbf{L}_{\mathrm{e},[a,b]}^{\mbf{n}_w}$.

\emph{Dual side.} $\mbf{D}(\Theta)$ has realization
\begin{equation}\label{eq:dual-filter-realization-proof}
\bmat{
\mcl T_\Theta^*\dot{\underline{\mbf{x}}}_\Theta(t)\\
\tilde{\underline z}(t)\\
\tilde{\underline w}(t)
}
=
\bmat{
\mcl A_\Theta^*
-
\bmat{-\mcl C_\Theta^{w*}&\mcl C_\Theta^{z*}}
\underline{\mcl D}_\Theta^{-1}
\bmat{-\mcl B_\Theta^{w*}\\\mcl B_\Theta^{z*}}
&
\bmat{-\mcl C_\Theta^{w*}&\mcl C_\Theta^{z*}}
\underline{\mcl D}_\Theta^{-1}
\\
-
\underline{\mcl D}_\Theta^{-1}
\bmat{-\mcl B_\Theta^{w*}\\\mcl B_\Theta^{z*}}
&
\underline{\mcl D}_\Theta^{-1}
}
\bmat{
\underline{\mbf{x}}_\Theta(t)\\
\underline z(t)\\
\underline w(t)
}.
\end{equation}
Define the inverse of the dual feedthrough block
\begin{equation}\label{eq:dual-feedthrough-block}
\underline{\mcl D}_\Theta:=
\bmat{
\mcl D_\Theta^{w*}&-\mcl D_\Theta^{zw*}\\
-\mcl D_\Theta^{wz*}&\mcl D_\Theta^{z*}
}.
\end{equation}
Define the Schur complement,
\begin{align*}
\mcl S_\Theta
&:=\mcl D_\Theta^{z*}
-(-\mcl D_\Theta^{wz*})W_\Theta^*
(-\mcl D_\Theta^{zw*})\\
&=\mcl D_\Theta^{z*}
-\mcl D_\Theta^{wz*}W_\Theta^*
\mcl D_\Theta^{zw*}
=
\bigl(\mcl D_\Theta^z
-\mcl D_\Theta^{zw}W_\Theta\mcl D_\Theta^{wz}
\bigr)^*
=
\bigl(\mcl D_\Theta^{\dagger,z}\bigr)^* .
\end{align*}
The block inverse formula therefore gives
\begin{equation*}
\underline{\mcl D}_\Theta^{-1}
=
\bmat{
W_\Theta^*+
W_\Theta^*\mcl D_\Theta^{zw*}\mcl S_\Theta^{-1}
\mcl D_\Theta^{wz*}W_\Theta^*
&
W_\Theta^*\mcl D_\Theta^{zw*}\mcl S_\Theta^{-1}
\\
\mcl S_\Theta^{-1}\mcl D_\Theta^{wz*}W_\Theta^*
&
\mcl S_\Theta^{-1}
}=
\bmat{(\mcl D_\Theta^{\dagger,w\tilde w})^*
-(\mcl D_\Theta^{\dagger,z\tilde w})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*
&
(\mcl D_\Theta^{\dagger,z\tilde w})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}\\
-\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*
&
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
}.
\end{equation*}
The Potapov--Ginzburg transform of $\mbf D(\Theta)$ is obtained by
solving the $\tilde{\underline w}$ row of
\eqref{eq:dual-filter-realization-proof}. Using the first inverse identity,
the state-column contribution in that row is
\begin{align*}
\bmat{0&I}\underline{\mcl D}_\Theta^{-1}
\bmat{-\mcl B_\Theta^{w*}\\\mcl B_\Theta^{z*}}
&=
-\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*
(-\mcl B_\Theta^{w*})
+
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\mcl B_\Theta^{z*}\\
&=
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\left(
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*\mcl B_\Theta^{w*}
+\mcl B_\Theta^{z*}
\right)\\
&=
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^* .
\end{align*}
Therefore
\begin{align*}
\tilde{\underline w}
&=
-\bmat{0&I}\underline{\mcl D}_\Theta^{-1}
\bmat{-\mcl B_\Theta^{w*}\\\mcl B_\Theta^{z*}}
\underline{\mbf{x}}_\Theta
+
\bmat{0&I}\underline{\mcl D}_\Theta^{-1}
\bmat{\underline z\\\underline w}\\
&=
-\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^*
\underline{\mbf{x}}_\Theta
-
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*\underline z
+
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}\underline w .
\end{align*}
Multiplication by $(\mcl D_\Theta^{\dagger,z})^*$ and rearrangement give
\begin{equation}\label{eq:dual-pg-solve-w}
\underline w
=
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^*
\underline{\mbf{x}}_\Theta
+
\bigl(\mcl D_\Theta^{\dagger,z}\bigr)^*
\tilde{\underline w}
+
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*\underline z .
\end{equation}
The two remaining state-column products are
\begin{align*}
\bmat{-\mcl C_\Theta^{w*}&\mcl C_\Theta^{z*}}
\underline{\mcl D}_\Theta^{-1}
\bmat{-\mcl B_\Theta^{w*}\\\mcl B_\Theta^{z*}}
&=
-\left(
(\mcl C_\Theta^{\dagger,w})^*
-
(\mcl C_\Theta^{\dagger,z})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*
\right)\mcl B_\Theta^{w*}\\
&\quad{}+
(\mcl C_\Theta^{\dagger,z})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\mcl B_\Theta^{z*}\\
&=
-(\mcl C_\Theta^{\dagger,w})^*\mcl B_\Theta^{w*}
+
(\mcl C_\Theta^{\dagger,z})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^*,
\\[1mm]
\bmat{I&0}\underline{\mcl D}_\Theta^{-1}
\bmat{-\mcl B_\Theta^{w*}\\\mcl B_\Theta^{z*}}
&=
-\left(
(\mcl D_\Theta^{\dagger,w\tilde w})^*
-
(\mcl D_\Theta^{\dagger,z\tilde w})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*
\right)\mcl B_\Theta^{w*}\\
&\quad{}+
(\mcl D_\Theta^{\dagger,z\tilde w})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\mcl B_\Theta^{z*}\\
&=
-(\mcl D_\Theta^{\dagger,w\tilde w})^*\mcl B_\Theta^{w*}
+
(\mcl D_\Theta^{\dagger,z\tilde w})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^* .
\end{align*}
For the state row, the preceding product and the third inverse identity give
\begin{align*}
\mcl T_\Theta^*\dot{\underline{\mbf{x}}}_\Theta
&=
\left(
\mcl A_\Theta^*+
(\mcl C_\Theta^{\dagger,w})^*\mcl B_\Theta^{w*}
-
(\mcl C_\Theta^{\dagger,z})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^*
\right)\underline{\mbf{x}}_\Theta\\
&\quad+
\left(
(\mcl C_\Theta^{\dagger,w})^*
-
(\mcl C_\Theta^{\dagger,z})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*
\right)\underline z
+
(\mcl C_\Theta^{\dagger,z})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\underline w\\
&=
\left(
\mcl A_\Theta^*+
(\mcl C_\Theta^{\dagger,w})^*\mcl B_\Theta^{w*}
-
(\mcl C_\Theta^{\dagger,z})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^*
\right)\underline{\mbf{x}}_\Theta\\
&\quad+
\left(
(\mcl C_\Theta^{\dagger,w})^*
-
(\mcl C_\Theta^{\dagger,z})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*
\right)\underline z\\
&\quad{}+
(\mcl C_\Theta^{\dagger,z})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\left(
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^*
\underline{\mbf{x}}_\Theta
+
\bigl(\mcl D_\Theta^{\dagger,z}\bigr)^*
\tilde{\underline w}
+
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*\underline z
\right)\\
&=
\left(
\mcl A_\Theta^*+
(\mcl C_\Theta^{\dagger,w})^*\mcl B_\Theta^{w*}
\right)\underline{\mbf{x}}_\Theta
+
\bigl(\mcl C_\Theta^{\dagger,z}\bigr)^*
\tilde{\underline w}
+
\bigl(\mcl C_\Theta^{\dagger,w}\bigr)^*
\underline z\\
&=
\bigl(\mcl A_\Theta^\dagger\bigr)^*
\underline{\mbf{x}}_\Theta
+
\bigl(\mcl C_\Theta^{\dagger,z}\bigr)^*
\tilde{\underline w}
+
\bigl(\mcl C_\Theta^{\dagger,w}\bigr)^*
\underline z .
\end{align*}
Similarly, the $\tilde{\underline z}$ row and the second inverse identity give
\begin{align*}
\tilde{\underline z}
&=
\left(
(\mcl D_\Theta^{\dagger,w\tilde w})^*\mcl B_\Theta^{w*}
-
(\mcl D_\Theta^{\dagger,z\tilde w})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}

&\quad+
\left(

-

\right)\underline z
+
(\mcl D_\Theta^{\dagger,z\tilde w})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\underline w\\
&=
\left(
(\mcl D_\Theta^{\dagger,w\tilde w})^*\mcl B_\Theta^{w*}
-
(\mcl D_\Theta^{\dagger,z\tilde w})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^*
\right)\underline{\mbf{x}}_\Theta\\
&\quad+
\left(
(\mcl D_\Theta^{\dagger,w\tilde w})^*
-
(\mcl D_\Theta^{\dagger,z\tilde w})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*
\right)\underline z\\
&\quad{}+
(\mcl D_\Theta^{\dagger,z\tilde w})^*
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\left(
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^*
\underline{\mbf{x}}_\Theta
+
\bigl(\mcl D_\Theta^{\dagger,z}\bigr)^*
\tilde{\underline w}
+
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*\underline z
\right)\\
&=
(\mcl D_\Theta^{\dagger,w\tilde w})^*
\mcl B_\Theta^{w*}\underline{\mbf{x}}_\Theta
+
\bigl(\mcl D_\Theta^{\dagger,z\tilde w}\bigr)^*
\tilde{\underline w}
+
(\mcl D_\Theta^{\dagger,w\tilde w})^*
\underline z\\
&=
\bigl(\mcl B_\Theta^{\dagger,\tilde w}\bigr)^*
\underline{\mbf{x}}_\Theta
+
\bigl(\mcl D_\Theta^{\dagger,z\tilde w}\bigr)^*
\tilde{\underline w}
+
\bigl(\mcl D_\Theta^{\dagger,w\tilde w}\bigr)^*
\underline z .
\end{align*}
Collecting these three rows gives the Potapov--Ginzburg realization of the
dual filter,
\begin{equation}\label{eq:dual-potapov-ginzburg-realization}
\bmat{\mcl{T}_{\Theta}^*\dot{\underline{\mbf{x}}}_{\Theta}(t)\\
\tilde{\underline{z}}(t)\\
\underline{w}(t)}
=
\bmat{
\bigl(\mcl{A}_{\Theta}^\dagger\bigr)^*
&
\bigl(\mcl{C}_{\Theta}^{\dagger,z}\bigr)^*
&
\bigl(\mcl{C}_{\Theta}^{\dagger,w}\bigr)^*\\
\bigl(\mcl{B}_{\Theta}^{\dagger,\tilde{w}}\bigr)^*
&
\bigl(\mcl{D}_{\Theta}^{\dagger,z\tilde{w}}\bigr)^*
&
\bigl(\mcl{D}_{\Theta}^{\dagger,w\tilde{w}}\bigr)^*\\
\bigl(\mcl{B}_{\Theta}^{\dagger,z}\bigr)^*
&
\bigl(\mcl{D}_{\Theta}^{\dagger,z}\bigr)^*
&
\bigl(\mcl{D}_{\Theta}^{\dagger,wz}\bigr)^*}
\bmat{\underline{\mbf{x}}_{\Theta}(t)\\
\tilde{\underline{w}}(t)\\
\underline{z}(t)}.
\end{equation}
Augmenting \eqref{eq:dual-potapov-ginzburg-realization} with $P^\top$, define
\begin{equation*}
\underline{\Lambda}_c:=
\left(
\bigl((\mcl D_\Theta^{\dagger,z}\bigr)^*\bigr)^{-1}
\left(I-\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*\mcl D^*\right)
\right)^{-1}.
\end{equation*}
The congruence transformations, $\underline{\Gamma}$ and $\underline{\Gamma}^{-1}$, respectively, which recover the direct dual systems are
\begin{equation}\label{eq:dual-pg-congruence-transform}
\begin{aligned}
\bmat{
\underline{\mbf{x}}(t)\\
\underline{\mbf{x}}_\Theta(t)\\
\tilde{\underline w}(t)
}
&=
\bmat{
I&0&0\\
0&I&0\\
\!\!-\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*\mcl B^*
&
\!\!-\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^*
&
\underline{\Lambda}_c^{-1}
}
\bmat{
\underline{\mbf{x}}(t)\\
\underline{\mbf{x}}_\Theta(t)\\
\underline w(t)
},
\\
\bmat{
\underline{\mbf{x}}(t)\\
\underline{\mbf{x}}_\Theta(t)\\
\underline w(t)
}
&=
\bmat{
I&0&0\\
0&I&0\\
\underline{\Lambda}_c
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl D_\Theta^{\dagger,wz}\bigr)^*\mcl B^*
&
\underline{\Lambda}_c
\bigl((\mcl D_\Theta^{\dagger,z})^*\bigr)^{-1}
\bigl(\mcl B_\Theta^{\dagger,z}\bigr)^*
&
\underline{\Lambda}_c
}
\bmat{
\underline{\mbf{x}}(t)\\
\underline{\mbf{x}}_\Theta(t)\\
\tilde{\underline w}(t)
}.
\end{aligned}
\end{equation}
The well-posedness of $P^\top\star(\Theta^\dagger)^\top$ gives the required
invertibility of
$I-(\mcl{D}_\Theta^{\dagger,wz})^*\mcl{D}^*$ which is satisfied iff the invertibility condition of the primal system is satisfied. 
$\bigl(\mcl D_\Theta^{\dagger,z}\bigr)^*$ is invertible by definition and Schur complement. Hence
$\underline{\Lambda}_c \in \Pi_4$ is well defined. Thus we find that
$\mbf{D}(\Theta)^\dagger = (\Theta^\dagger)^\top$ and
$\mbf{D}(\Theta)\bmat{P^\top\\ I}\underline{w} = \bmat{P^\top \star (\Theta^\dagger)^\top \\ I}\underline{\Gamma} \underline{w}$
for all $\underline{w}\in\mbf{L}_{\mathrm{e},[a,b]}^{\mbf{n}_z}$.
