In this work we assume that the generalized plant $P$ is a PIE system, that admits a state-space representation.
This work builds on the duality results in \cite{shivaknew} which shows that the state-feedback controller results in a convex optimization problem.
We find that this duality result can be extended to the case of dynamic filters, where we want to know the properties of the dual dynamic filter. This is particularly
usefull when examining the effect of uncertainties on the system when performing state-feedback controller synthesis, when using IQCs. The works from
Seiler and Venkataraman \cite{6915700,venkataraman} show that, after $J$-spectral factorization, the dual of a dynamic filter can be found using the Potapov-Ginzburg transform.
This section extends these results to PIE systems, and shows that the dual of a PIE system with a dynamic filter can be found
using the Potapov-Ginzburg transform, identical to the work from Venkataraman. 

\subsection{Duality under Dynamical Filtering}
We begin by restating the main theorem from \cite{shivaknew}.

\begin{theorem}[Dual $L_2$-gain
{\cite[Thm.~9]{shivaknew}}]\label{thm:dual-l2-gain}
For a PIE system with corresponding PI operators $P\{\mcl{T}, \mcl{A}, \mcl{B},
\mcl{C}, \mcl{D}\}\in\Hinf$, the following statements are equivalent.
\begin{enumerate}
\item For any $w\in\mbf{L}_{\mathrm{e},[a,b]}^{\mbf{n}_w}$, and $\mbf{x}(0)=0$, $P$
satisfies $\|p_T {z}\| \leq \|p_T {w}\|$.
\item For any $\underline{w}\in\mbf{L}_{\mathrm{e},[a,b]}^{\mbf{n}_z}$, and
$\underline{\mbf{x}}(0)=0$, $P^\top$ satisfies $\|p_T \underline{z}\| \leq
\|p_T \underline{w}\|$.
\end{enumerate}
\end{theorem}
Relating an IQC to this gain duality requires two transformations. First,
write $\bmat{\tilde z\\\tilde w}:=\Theta\bmat{z\\w}$. A $J$-spectral
factorization gives, for every $T\geq0$,
\begin{equation*}
\begin{aligned}
&\ip{p_T\Psi\bmat{z\\w}}
{p_T\mcl M_{\mcl V}\Psi\bmat{z\\w}}_{\mbf L_{[a,b]}}\\
&\quad=
\|p_T\tilde z\|_{\mbf L_{[a,b]}}^2
-\|p_T\tilde w\|_{\mbf L_{[a,b]}}^2.
\end{aligned}
\end{equation*}
Thus the IQC quadratic form becomes the difference between the filtered
output and input signals. 

\begin{definition}[$J$-Spectral Factorization and Dual Filter
{\cite{venkataraman}}]
\label{def:dynamic-filters}\label{def:factorization}
Let $(\Psi,\mcl M_{\mcl V})$ be a hard IQC, let
\begin{equation*}
J_{\mbf n_z,\mbf n_w}:=
\bmat{I_{\mbf n_z}&0\\0&-I_{\mbf n_w}},
\end{equation*}
and let $\Theta,\Theta^{-1}, \Theta_{22}, \Theta_{22}^{-1}$ be causal bounded linear operators. The pair
$(\Theta,J_{\mbf n_z,\mbf n_w})$ is a $J$-spectral factorization of
$(\Psi,\mcl M_{\mcl V})$ if,
\begin{equation*}
\begin{aligned}
&
\ip{p_T\Psi\bmat{z\\w}}
{p_T\mcl M_{\mcl V}\Psi\bmat{z\\w}}_{\mbf L_{[a,b]}}
\\[-1mm]
&\quad=
\ip{p_T\Theta\bmat{z\\w}}
{p_TJ_{\mbf n_z,\mbf n_w}\Theta\bmat{z\\w}}_{\mbf L_{[a,b]}}.
\end{aligned}
\end{equation*}
Define the matrix
$\hat{J}:=\bmat{0_{\mbf n_z\times\mbf n_w}&I_{\mbf n_z}\\-I_{\mbf n_w}&0_{\mbf n_w\times\mbf n_z}}$. The dual filter associated with $\Theta$ is
defined by
\begin{equation*}
\mbf{D}(\Theta)
:=
\left(\hat{J}^*\Theta^\top\hat{J}\right)^{-1}.
\end{equation*}
\end{definition}
Finding a $J$--spectral factorization is the first step in applying the duality theory. For the construction of a $J$-spectral factorization, we partition
\begin{equation*}
\Theta=\bmat{\Theta_z & \Theta_{zw}\\ \Theta_{wz} & \Theta_w}.
\end{equation*}
The associated primal and dual filtered signals are
\begin{equation*}
\Theta\bmat{z(t) \\ w(t)}
=
\bmat{\tilde{z}(t) \\ \tilde{w}(t)}, \qquad \mbf{D}(\Theta)
\bmat{\underline{z}(t) \\ \underline{w}(t)}
=
\bmat{\tilde{\underline{z}}(t) \\ \tilde{\underline{w}}(t)}.
\end{equation*}
The $J$-spectral factorization for the primal IQC leads to the state-space realization
\begin{equation*}
\bmat{\mcl{T}_{\Theta}\dot{\mbf{x}}_{\Theta}(t)\\ \tilde{z}(t) \\ \tilde{w}(t)}
=
\bmat{\mcl{A}_{\Theta} & \mcl{B}_{\Theta}^z & \mcl{B}_{\Theta}^w \\
\mcl{C}_{\Theta}^z & \mcl{D}_{\Theta}^z & \mcl{D}_{\Theta}^{zw}\\
\mcl{C}_{\Theta}^w & \mcl{D}_{\Theta}^{wz} & \mcl{D}_{\Theta}^w}
\bmat{\mbf{x}_{\Theta}(t) \\ z(t) \\ w(t)},
\end{equation*}
where the entries are PI operators of appropriate dimension.
The second step is the required graph rearrangement, given by the Potapov--Ginzburg transform. We use the following
construction from \cite{venkataraman}.

\begin{definition}(Potapov--Ginzburg Transform)\label{def:potapov-ginzburg-transform}
Given $\Theta:\bmat{z\\w} \to \bmat{\tilde{z}\\\tilde{w}}$ as in Definition~\ref{def:dynamic-filters}, the Potapov--Ginzburg transform denoted
by $\Theta^\dagger:\bmat{z\\\tilde{w}} \to \bmat{\tilde{z}\\w}$ is defined by
\begin{equation*}
\Theta^\dagger
=
\bmat{
\Theta_z-\Theta_{zw}\Theta_w^{-1}\Theta_{wz}
&
\Theta_{zw}\Theta_w^{-1}\\
-\Theta_w^{-1}\Theta_{wz}
&
\Theta_w^{-1}
},
\end{equation*}
\end{definition}
The transform identifies the filtered graph of $P$ with the graph of
$P\star(\Theta^\dagger R)$, under an invertible change of coordinates. The next
lemma establishes this relation for both the primal and dual
interconnections.

\begin{lemma} (Potapov--Ginzburg Transform Equivalence)\label{lem:potapov-ginzburg-equivalence}
Let $\Theta, \mbf{D}(\Theta)\in \Hinf$ as in Definition~\ref{def:dynamic-filters}. Moreover, let $(P\star(\Theta^\dagger R))\in\mcl{G}_2$
and $(P^\top\star(R(\Theta^\dagger)^\top))\in\mcl{G}_2$ be well-posed PIE systems, such that
\[
(\mcl{D}_{\Theta}^{wz} \mcl{D} + \mcl{D}_{\Theta}^w)^{-1},
\]
exist and are closed under inversion. Then, there exists invertible coordinate transformations $\mcl{M}_{\Gamma},\mcl{M}_{\Gamma^{-1}}\in\Pi_4$ on the primal side and
$\mcl{M}_{\underline{\Gamma}},\mcl{M}_{\underline{\Gamma}^{-1}}\in\Pi_4$ on the dual side such that
\begin{itemize}
    \item $\Theta\bmat{P\\I} w = \bmat{P\star(\Theta^\dagger R) \\ I}\mcl{M}_{\Gamma} w$ for all $w\in\mbf{L}_{\mathrm{e},[a,b]}^{\mbf{n}_w}$.
    \item $\mbf{D}(\Theta)\bmat{P^\top\\I} \underline{w} = \bmat{P^\top\star(R(\Theta^\dagger)^\top) \\ I}\mcl{M}_{\underline{\Gamma}}\,\underline{w}$ for all $\underline{w}\in\mbf{L}_{\mathrm{e},[a,b]}^{\mbf{n}_z}$.
\end{itemize}
\end{lemma}
