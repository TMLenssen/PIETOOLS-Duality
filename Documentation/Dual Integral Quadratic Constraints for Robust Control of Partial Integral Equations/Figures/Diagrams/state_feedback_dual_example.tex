% Reusable controller-interconnection schematics for the component boxes.
% Each command produces only a TikZ picture, so it can be positioned by the
% surrounding layout without creating a separate numbered figure.

\tikzset{
    controller schematic block/.style={
        draw,
        minimum width=1cm,
        minimum height=1cm,
        align=center,
        inner sep=1pt,
        line width=0.5pt
    },
    controller schematic controller/.style={
        draw, dashed,
        minimum width=0.72cm,
        minimum height=0.72cm,
        align=center,
        inner sep=1pt,
        line width=0.5pt
    },
    controller schematic uncertainty/.style={
        draw,
        dashed,
        minimum width=0.6cm,
        minimum height=0.6cm,
        align=center,
        inner sep=1pt,
        line width=0.5pt
    },
    controller schematic label/.style={
        fill=white,
        inner sep=0.7pt,
        outer sep=1pt
    },
    controller schematic signal/.style={
        ->,
        line width=0.5pt
    }
}
